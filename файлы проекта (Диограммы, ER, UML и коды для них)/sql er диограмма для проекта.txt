// Пользователь
table user {
  id SERIAL [NOT NULL, PK]
  username VARCHAR(50) [NOT NULL]
  user_email VARCHAR [NOT NULL, unique]
  user_password TEXT [NOT NULL]
  created_at timestamptz(CURRENT_TIMESTAMPTZ)
}

// администратор
table admin {
  id SERIAL [NOT NULL, PK]
  username VARCAHR(100) [NOT NULL, unique]
  password PASSWORD_HASH [not null]
  created_at timestamptz(CURRENT_TIMESTAMPTZ)
}

// Приложения
table application {
  id SERIAL [not null, PK]
  title VARCHAR(100) [NOT NULL]
  created_at timestamptz(CURRENT_TIMESTAMPTZ)
}

// Ключи
table key {
  id SERIAL [NOT NULL, PK]
  key_kode VARCHAR(50) [NOT NULL, unique]
  key_duration INT [NOT NULL]
  key_state BOOLEAN(false) [NOT NULL]
  activated_at timestamptz
  expires_at timestamptz
  created_by INT [ref: >? admin.id]
  application_id INT [ref: >? application.id]
}

// Заказы
table order {
  id SERIAL [NOT NULL, PK]
  user_id INT [ref: >? user.id]
  status VARCHAR(30)
  total_amount DECIMAL(5,0) [NOT NULL]
  payment_method VARCHAR(30) [NOT NULL]
  created_at timestamptz(CURRENT_TIMESTAMPTZ)
}

// Купленные ключи
table order_keys {  
  id SERIAL [NOT NULL, PK]
  user_id INT [ref: >? user.id]
  order_id INT [ref: >? order.id]
  key_id INT [ref: >? key.id]
  duration_days INT [NOT NULL]
  purchase_date timestamptz(CURRENT_TIMESTAMPTZ)
}

// ключи пользователей
table user_key {
  id SERIAL [NOT NULL, PK]
  user_id INT [ref: ?>? user.id]
  key_id INT [ref: >? key.id]
  activation_date timestamptz()
  end_date timestamptz
}
